                        <span class="text-xs font-medium px-2 py-1 rounded-full ${categoryClass}">${item.category === 'Sightseeing' ? '景點' : item.category === 'Food' ? '吃飯' : item.category === 'Shopping' ? '逛街' : item.category === 'Travel' ? '交通' : item.category === 'Accom' ? '住宿' : '其他'}</span>
                    </div>
                    <p class="text-base text-gray-700 font-medium">${item.desc}</p>
                    <div class="flex items-center space-x-4 text-sm text-gray-500 pt-1">
                        ${item.notes ? `<span class="flex items-center"><i data-lucide="notebook-pen" class="icon w-4 h-4 mr-1"></i>${item.notes}</span>` : ''}
                        ${item.map ? 
                            // 地圖連結顏色優化為藍色
                            `<a href="${item.map}" target="_blank" class="flex items-center text-white bg-blue-500 hover:bg-blue-600 rounded-full px-2 py-0.5 text-xs font-semibold transition-colors shadow-sm">
                                <i data-lucide="map-pin" class="icon w-3 h-3 mr-1"></i>
                                Google 地圖
                            </a>` : ''}
                    </div>
                </div>
                <div class="w-6 text-gray-400 hover:text-blue-500 transition-colors pt-2">
                    <i data-lucide="chevron-right" class="icon w-5 h-5"></i>
                </div>
            `;
            return itemElement;
        };


        // --- B. 拖曳與排序 (Drag and Drop) 邏輯 ---
        let dragSrcEl = null;

        const attachItemEventListeners = () => {
            document.querySelectorAll('.itinerary-item').forEach(item => {
                // 1. 點擊編輯
                item.addEventListener('click', (e) => {
                    // 避免點擊地圖連結時觸發編輯
                    if (e.target.closest('a')) return; 
                    openEditModal(item.dataset.dayKey, item.dataset.id)
                });
                
                // 2. 拖曳事件
                item.addEventListener('dragstart', handleDragStart);
                item.addEventListener('dragover', handleDragOver);
                item.addEventListener('dragleave', handleDragLeave);
                item.addEventListener('drop', handleDrop);
                item.addEventListener('dragend', handleDragEnd);
            });
        };

        function handleDragStart(e) {
            dragSrcEl = this;
            e.dataTransfer.effectAllowed = 'move';
            e.dataTransfer.setData('text/html', this.outerHTML);
            this.classList.add('dragging');
        }

        function handleDragOver(e) {
            e.preventDefault(); // 允許拖曳
            e.dataTransfer.dropEffect = 'move';

            if (dragSrcEl !== this) {
                // 視覺提示：判斷是在上方還是下方
                const rect = this.getBoundingClientRect();
                const offsetY = e.clientY - rect.top;
                
                document.querySelectorAll('.itinerary-item').forEach(el => {
                    el.classList.remove('drag-over-top', 'drag-over-bottom');
                });

                if (offsetY < rect.height / 2) {
                    this.classList.add('drag-over-top');
                } else {
                    this.classList.add('drag-over-bottom');
                }
            }
        }

        function handleDragLeave(e) {
            this.classList.remove('drag-over-top', 'drag-over-bottom');
        }

        function handleDrop(e) {
            e.stopPropagation();
            this.classList.remove('drag-over-top', 'drag-over-bottom');

            if (dragSrcEl !== this) {
                const sourceDayKey = dragSrcEl.dataset.dayKey;
                const targetDayKey = this.dataset.dayKey;
                const sourceId = dragSrcEl.dataset.id;
                const targetId = this.dataset.id;
                const sourceList = window.itineraryData[sourceDayKey];
                const targetList = window.itineraryData[targetDayKey];

                const sourceIndex = sourceList.findIndex(item => item.id === sourceId);
                const targetIndex = targetList.findIndex(item => item.id === targetId);
                
                // 1. 移除拖曳的項目
                const draggedItem = sourceList.splice(sourceIndex, 1)[0];
                
                // 2. 決定插入的位置
                const rect = this.getBoundingClientRect();
                const offsetY = e.clientY - rect.top;
                
                let insertIndex = targetIndex;
                if (offsetY > rect.height / 2) {
                    insertIndex = targetIndex + 1;
                }
                
                let itemToInsert = draggedItem;

                if (sourceDayKey !== targetDayKey) {
                    // 跨日拖曳：賦予新的唯一 ID 
                    itemToInsert = { ...draggedItem, id: window.generateNewId(targetDayKey) };
                }

                // 插入到目標列表
                targetList.splice(insertIndex, 0, itemToInsert);
                
                // 3. 更新數據庫 (Mock Mode: 更新本地資料並重新渲染)
                const newData = { ...window.itineraryData };
                window.updateItineraryInDB(newData);
            }
        }

        function handleDragEnd(e) {
            this.classList.remove('dragging', 'drag-over-top', 'drag-over-bottom');
            dragSrcEl = null;
        }

        // --- C. Modal (彈出視窗) 編輯邏輯 ---

        const modal = document.getElementById('edit-modal');
        const modalContent = document.getElementById('modal-content');
        const daySelect = document.getElementById('edit-day');

        // 打開 Modal
        const openModal = (dayKey = null, itemId = null) => {
            editingItem = null;
            const form = document.getElementById('itinerary-form');
            form.reset();
            document.getElementById('delete-button').classList.add('hidden');
            document.getElementById('modal-title').textContent = '新增行程項目';
            document.getElementById('day-select-group').classList.remove('hidden'); // 預設顯示日期選擇
            daySelect.innerHTML = ''; // 清空選項 (新增時重新填入)

            if (itemId) {
                // 編輯模式
                const dayList = window.itineraryData[dayKey];
                editingItem = dayList.find(item => item.id === itemId);
                if (!editingItem) return;

                document.getElementById('modal-title').textContent = '編輯行程';
                document.getElementById('edit-day-key').value = dayKey;
                document.getElementById('edit-item-id').value = itemId;
                document.getElementById('edit-time').value = editingItem.time;
                document.getElementById('edit-desc').value = editingItem.desc;
                document.getElementById('edit-category').value = editingItem.category || 'Other';
                document.getElementById('edit-notes').value = editingItem.notes || '';
                document.getElementById('edit-map').value = editingItem.map || '';
                
                document.getElementById('delete-button').classList.remove('hidden');
                document.getElementById('day-select-group').classList.add('hidden'); // 編輯時不允許跨日選擇
            } else if (dayKey) {
                // 點擊當日新增按鈕，預設選擇當日
                document.getElementById('edit-day-key').value = dayKey;
                document.getElementById('day-select-group').classList.add('hidden');
            } else {
                 // 點擊右下角 '+' 時，強制選擇日期
                 Object.keys(DAYS_MAP).forEach(key => {
                    daySelect.innerHTML += `<option value="${key}">${key} (${DAYS_MAP[key]})</option>`;
                });
            }

            modal.classList.remove('hidden');
            modal.classList.add('flex');
            setTimeout(() => {
                modalContent.classList.remove('opacity-0', 'scale-95');
            }, 10);
            lucide.createIcons(); // 確保 Modal 內的 Icon 渲染
        };

        // 關閉 Modal
        const closeEditModal = () => {
            modalContent.classList.add('opacity-0', 'scale-95');
            setTimeout(() => {
                modal.classList.add('hidden');
                modal.classList.remove('flex');
                daySelect.innerHTML = ''; // 清空選項
            }, 300);
        };

        // 編輯單個行程項目
        const openEditModal = (dayKey, itemId) => {
            openModal(dayKey, itemId);
        };

        // --- D. 表單提交與數據更新 ---

        document.getElementById('itinerary-form').addEventListener('submit', (e) => {
            e.preventDefault();

            const formData = new FormData(e.target);
            const itemId = formData.get('item');
            let dayKey = formData.get('day') || formData.get('edit-day-key');
            const currentId = formData.get('edit-item-id');

            const newItem = {
                id: currentId || window.generateNewId(dayKey),
                time: formData.get('time'),
                desc: formData.get('desc'),
                notes: formData.get('notes'),
                category: formData.get('category'),
                map: formData.get('map'),
            };

            const newData = { ...window.itineraryData };
            
            if (currentId) {
                // 編輯現有項目
                const dayList = newData[dayKey];
                const index = dayList.findIndex(item => item.id === currentId);
                if (index !== -1) {
                    dayList[index] = newItem;
                }
            } else {
                // 新增項目
                if (!newData[dayKey]) newData[dayKey] = [];
                // 簡單插入
                newData[dayKey].push(newItem);
                // 嘗試按時間排序
                newData[dayKey].sort((a, b) => a.time.localeCompare(b.time));
            }
            
            window.updateItineraryInDB(newData);
            closeEditModal();
        });

        // 刪除行程 (Unchanged)
        document.getElementById('delete-button').addEventListener('click', () => {
            if (!editingItem) return;

            // 用 custom modal 替代 alert
            const confirmDelete = window.confirm || ((message) => {
                console.warn("Using console fallback for confirmation:", message);
                return true; // Auto-confirm in non-browser envs
            });

            if (!confirmDelete("確定要刪除這個行程項目嗎？此操作無法恢復。")) return;

            const dayKey = document.getElementById('edit-day-key').value;
            const itemId = document.getElementById('edit-item-id').value;

            if (dayKey && itemId) {
                const newData = { ...window.itineraryData };
                newData[dayKey] = newData[dayKey].filter(item => item.id !== itemId);
                window.updateItineraryInDB(newData);
                closeEditModal();
            }
        });

        // 關閉 Modal 按鈕 (Unchanged)
        document.getElementById('close-modal-button').addEventListener('click', closeEditModal);
        modal.addEventListener('click', (e) => {
            if (e.target === modal) {
                closeEditModal();
            }
        });

        // 右下角新增按鈕 (Unchanged)
        document.getElementById('add-button').addEventListener('click', () => openModal(null, null));

        // --- E. 輔助功能：即時時間/天氣模擬 (Unchanged) ---
        function updateClock() {
            const now = new Date();
            const timeStr = now.toLocaleTimeString('zh-TW', { hour12: false });
            document.getElementById('current-time').textContent = timeStr;
            
            // 簡化模擬天氣建議 (基於沖繩3月氣候)
            const weatherAdvice = `平均 20°C，建議多層次穿搭。`;
            document.getElementById('current-weather').textContent = weatherAdvice;
        }
        setInterval(updateClock, 1000);
        updateClock(); // 立即執行一次

        // --- F. 新增當日行程按鈕 (監聽新增的日區塊按鈕) ---
        // 點擊每個 Day Header 下方的按鈕來新增該日行程
        document.addEventListener('click', (e) => {
            // 使用 closest 查找 add-day-item 類別的父元素
            const addButton = e.target.closest('.add-day-item');
            if (addButton) {
                const dayKey = addButton.dataset.dayKey;
                openModal(dayKey, null);
            }
        });
